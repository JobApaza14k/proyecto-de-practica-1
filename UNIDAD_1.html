Trabajo 1: Aproximación de Raíces

Método de Newton

Implementación del método de Newton-Raphson para la aproximación de raíces de funciones no lineales.

import numpy as np

def newton_raphson(f, df, x0, tol=1e-6, max_iter=100):
    """
    Método de Newton-Raphson
    f: función
    df: derivada de la función
    x0: aproximación inicial
    """
    x = x0
    for i in range(max_iter):
        fx = f(x)
        if abs(fx) < tol:
            print(f"Convergió en {i} iteraciones")
            return x
        
        dfx = df(x)
        if dfx == 0:
            print("Derivada cero. No se puede continuar.")
            return None
        
        x = x - fx / dfx
    
    print("No convergió en el máximo de iteraciones")
    return x

# Ejemplo: encontrar raíz de x^2 - 4
f = lambda x: x**2 - 4
df = lambda x: 2*x

raiz = newton_raphson(f, df, 1.0)
print(f"Raíz encontrada: {raiz}")

Trabajo 2: Interpolación Polinomial

Lagrange

Implementación del método de interpolación polinomial de Lagrange para aproximar valores intermedios.

import numpy as np
import matplotlib.pyplot as plt

def interpolacion_lagrange(x, y, xi):
    """
    Interpolación polinomial de Lagrange
    x: puntos conocidos (x)
    y: valores conocidos (y)
    xi: punto a interpolar
    """
    n = len(x)
    resultado = 0
    
    for i in range(n):
        termino = y[i]
        for j in range(n):
            if i != j:
                termino *= (xi - x[j]) / (x[i] - x[j])
        resultado += termino
    
    return resultado

# Ejemplo
x = np.array([0, 1, 2, 3])
y = np.array([1, 3, 2, 5])
xi = 1.5

yi = interpolacion_lagrange(x, y, xi)
print(f"Valor interpolado en x={xi}: {yi}")

Trabajo 3: Eliminación Gaussiana

Sistemas Lineales

Resolución de sistemas de ecuaciones lineales mediante el método de eliminación gaussiana con pivoteo.

import numpy as np

def eliminacion_gaussiana(A, b):
    """
    Resuelve Ax = b usando eliminación gaussiana
    """
    n = len(b)
    Ab = np.column_stack([A, b]).astype(float)
    
    # Eliminación hacia adelante
    for i in range(n):
        # Pivoteo parcial
        max_row = i + np.argmax(np.abs(Ab[i:, i]))
        Ab[[i, max_row]] = Ab[[max_row, i]]
        
        for j in range(i+1, n):
            factor = Ab[j, i] / Ab[i, i]
            Ab[j, i:] -= factor * Ab[i, i:]
    
    # Sustitución hacia atrás
    x = np.zeros(n)
    for i in range(n-1, -1, -1):
        x[i] = (Ab[i, -1] - np.dot(Ab[i, i+1:n], x[i+1:])) / Ab[i, i]
    
    return x

# Ejemplo
A = np.array([[2, 1, -1], [-3, -1, 2], [-2, 1, 2]])
b = np.array([8, -11, -3])
solucion = eliminacion_gaussiana(A, b)
print("Solución:", solucion)
